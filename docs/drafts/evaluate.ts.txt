import { readFileSync } from 'node:fs';
import { PassageIndex, concepts, expandedQuery, mmr } from '../src/core-ir.js';
import type { Branch, Passage, SearchHit } from '../src/types.js';

interface JudgedQuery { id: string; text: string; judgments: Record<string, number>; }
interface Dataset { passages: Passage[]; queries: JudgedQuery[]; }
interface Metrics { p5: number; p10: number; ndcg10: number; pooledRecall10: number; redundancy10: number; }

function parseDataset(value: unknown): Dataset {
  if (!value || typeof value !== 'object') throw new Error('Dataset must be an object');
  const dataset = value as Record<string, unknown>;
  if (!Array.isArray(dataset.passages) || !Array.isArray(dataset.queries)) throw new Error('Expected passages[] and queries[]');
  const ids = new Set<string>();
  const passages = dataset.passages.map(raw => {
    if (!raw || typeof raw !== 'object') throw new Error('Invalid passage');
    const p = raw as Record<string, unknown>;
    if (typeof p.id !== 'string' || ids.has(p.id) || typeof p.snapshotId !== 'string' || typeof p.text !== 'string' || typeof p.duplicateGroup !== 'string' || !Number.isInteger(p.start) || !Number.isInteger(p.end) || Number(p.start) < 0 || Number(p.end) - Number(p.start) !== p.text.length) throw new Error('Passages require unique IDs, exact UTF-16 offsets and duplicate groups');
    ids.add(p.id);
    return { id: p.id, snapshotId: p.snapshotId, text: p.text, duplicateGroup: p.duplicateGroup, start: Number(p.start), end: Number(p.end) } satisfies Passage;
  });
  const queryIds = new Set<string>();
  const queries = dataset.queries.map(raw => {
    if (!raw || typeof raw !== 'object') throw new Error('Invalid query');
    const q = raw as Record<string, unknown>;
    if (typeof q.id !== 'string' || queryIds.has(q.id) || typeof q.text !== 'string' || !q.text.trim() || !q.judgments || typeof q.judgments !== 'object' || Array.isArray(q.judgments)) throw new Error('Queries require unique IDs, text and passage judgments');
    queryIds.add(q.id);
    const judgments = q.judgments as Record<string, unknown>;
    for (const [id, relevance] of Object.entries(judgments)) if (!ids.has(id) || typeof relevance !== 'number' || !Number.isInteger(relevance) || relevance < 0 || relevance > 3) throw new Error(`Invalid relevance judgment for ${id}`);
    return { id: q.id, text: q.text, judgments: judgments as Record<string, number> };
  });
  if (!passages.length || !queries.length) throw new Error('Empty evaluation corpus or query pool');
  return { passages, queries };
}

export function metrics(hits: SearchHit[], judgments: Record<string, number>): Metrics {
  const relevant = (h: SearchHit) => (judgments[h.passage.id] ?? 0) > 0;
  const top = hits.slice(0, 10);
  const dcg = (scores: number[]) => scores.reduce((sum, grade, i) => sum + (2 ** grade - 1) / Math.log2(i + 2), 0);
  const ideal = dcg(Object.values(judgments).sort((a, b) => b - a).slice(0, 10));
  const poolCount = Object.values(judgments).filter(v => v > 0).length;
  return {
    p5: hits.slice(0, 5).filter(relevant).length / 5,
    p10: top.filter(relevant).length / 10,
    ndcg10: ideal ? dcg(top.map(h => judgments[h.passage.id] ?? 0)) / ideal : 0,
    pooledRecall10: poolCount ? top.filter(relevant).length / poolCount : 0,
    redundancy10: top.length ? 1 - new Set(top.map(h => h.passage.duplicateGroup)).size / top.length : 0,
  };
}

export function evaluate(dataset: Dataset) {
  const index = new PassageIndex(); index.add(dataset.passages);
  const results = dataset.queries.map(query => {
    const branch: Branch = { id: query.id, objectiveId: query.id, question: query.text, anchor: query.text, origin: 'initial', introducing: [], priority: 1, status: 'active', needs: [], aliases: [], queries: [], zeroYieldEpisodes: 0, stats: {} };
    const base = index.rank(query.text, 30);
    // No qrel leakage: lexical PRF candidates depend only on ranked corpus text.
    // This is a retrieval-only ablation, not the live contextual model-assessment controller.
    const candidates = concepts(base, branch, []);
    const expanded = index.rank(expandedQuery(query.text, candidates), 30);
    return { queryId: query.id, bm25: metrics(base, query.judgments), anchoredLexicalPRF: metrics(expanded, query.judgments), prfMMR: metrics(mmr(expanded, 10), query.judgments), ranking: { bm25: base.slice(0, 10).map(h => h.passage.id), prf: expanded.slice(0, 10).map(h => h.passage.id) } };
  });
  return { scope: 'Frozen-corpus retrieval-only ablation; unjudged passages count as zero. Recall is pooled recall, not web recall. Lexical PRF is not a substitute for contextual assessment in the live engine.', corpusPassages: dataset.passages.length, queries: dataset.queries.length, results };
}

const filename = process.argv[2];
if (!filename) {
  console.error('Usage: npm run evaluate -- <judged-dataset.json>\nDataset: {passages: Passage[], queries:[{id,text,judgments:{passageId:0..3}}]}. Supply actual human judgments; no results are fabricated.');
  process.exitCode = 1;
} else {
  try {
    const value: unknown = JSON.parse(readFileSync(filename, 'utf8'));
    console.log(JSON.stringify(evaluate(parseDataset(value)), null, 2));
  } catch (error) {
    console.error(error instanceof Error ? error.message : String(error));
    process.exitCode = 1;
  }
}
